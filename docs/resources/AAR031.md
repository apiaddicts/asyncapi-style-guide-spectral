The examples of a message should match its `payload` and `headers` schemas, so that they illustrate real messages.

The rule checks every message in `components.messages`, including each alternative of a `oneOf`. A message without a `payload` is reported. For each entry of the message `examples`, the rule reports an example without a `payload`, an example without `headers` when the message defines `headers`, and an example `payload` with top-level keys that are not declared in the `properties` of the message `payload`.

Noncompliant Code Example (AsyncAPI 2)
--------------------------------------

```yaml
asyncapi: 2.6.0
info:
  title: Orders Events
  version: 1.0.0
channels:
  orders.created:
    publish:
      message:
        $ref: '#/components/messages/OrderCreated'
components:
  messages:
    OrderCreated:
      name: OrderCreated
      contentType: application/json
      headers:
        type: object
        properties:
          correlationId:
            type: string
      payload:
        type: object
        properties:
          orderId:
            type: string
          total:
            type: number
      examples:
        - name: orderWithDiscount # Noncompliant {{AAR031: Example at index 0 should include 'headers' matching the message headers schema.}}
          payload: # Noncompliant {{AAR031: Example at index 0 contains keys not defined in payload schema: discount.}}
            orderId: ORD-1001
            total: 59.9
            discount: 10
```

Compliant Solution (AsyncAPI 2)
-------------------------------

```yaml
asyncapi: 2.6.0
info:
  title: Orders Events
  version: 1.0.0
channels:
  orders.created:
    publish:
      message:
        $ref: '#/components/messages/OrderCreated'
components:
  messages:
    OrderCreated:
      name: OrderCreated
      contentType: application/json
      headers:
        type: object
        properties:
          correlationId:
            type: string
      payload:
        type: object
        properties:
          orderId:
            type: string
          total:
            type: number
      examples:
        - name: orderWithDiscount
          headers:
            correlationId: 7f3c2a10-5b1e-4c3d-9a8e-2f6b1d4e0c9a
          payload:
            orderId: ORD-1001
            total: 59.9
```

Noncompliant Code Example (AsyncAPI 3)
--------------------------------------

```yaml
asyncapi: 3.0.0
info:
  title: Orders Events
  version: 1.0.0
channels:
  orderCreated:
    address: orders.created
    messages:
      orderCreated:
        $ref: '#/components/messages/OrderCreated'
components:
  messages:
    OrderCreated:
      name: OrderCreated
      contentType: application/json
      payload:
        type: object
        properties:
          orderId:
            type: string
      examples:
        - name: missingPayload # Noncompliant {{AAR031: Example at index 0 should include a 'payload' matching the message schema.}}
          summary: An example without a payload.
    OrderCancelled: # Noncompliant {{AAR031: Message should define a 'payload' schema to describe the message content.}}
      name: OrderCancelled
      contentType: application/json
```

Compliant Solution (AsyncAPI 3)
-------------------------------

```yaml
asyncapi: 3.0.0
info:
  title: Orders Events
  version: 1.0.0
channels:
  orderCreated:
    address: orders.created
    messages:
      orderCreated:
        $ref: '#/components/messages/OrderCreated'
components:
  messages:
    OrderCreated:
      name: OrderCreated
      contentType: application/json
      payload:
        type: object
        properties:
          orderId:
            type: string
      examples:
        - name: newOrder
          summary: An order that has just been created.
          payload:
            orderId: ORD-1001
    OrderCancelled:
      name: OrderCancelled
      contentType: application/json
      payload:
        type: object
        properties:
          orderId:
            type: string
```
