Every message is recommended to have a human-readable `title`.

In AsyncAPI 2 the rule checks the `message` of every `publish` and `subscribe` operation in `channels` and every message in `components.messages`. In AsyncAPI 3 it checks every message in `components.messages` and every message listed in the `messages` of an operation in the root `operations` object. A message given as a `$ref` is reported where it is defined. A message whose `title` is missing or empty is reported.

Noncompliant Code Example (AsyncAPI 2)
--------------------------------------

```yaml
asyncapi: 2.6.0
info:
  title: Orders Events
  version: 1.0.0
channels:
  orders.created:
    publish:
      message: # Noncompliant {{AAR035: Message should have a 'title' field.}}
        name: OrderCreated
        contentType: application/json
        payload:
          type: object
  orders.cancelled:
    subscribe:
      message:
        $ref: '#/components/messages/OrderCancelled'
components:
  messages:
    OrderCancelled: # Noncompliant {{AAR035: Message should have a 'title' field.}}
      name: OrderCancelled
      contentType: application/json
      payload:
        type: object
```

Compliant Solution (AsyncAPI 2)
-------------------------------

```yaml
asyncapi: 2.6.0
info:
  title: Orders Events
  version: 1.0.0
channels:
  orders.created:
    publish:
      message:
        name: OrderCreated
        title: Order created
        contentType: application/json
        payload:
          type: object
  orders.cancelled:
    subscribe:
      message:
        $ref: '#/components/messages/OrderCancelled'
components:
  messages:
    OrderCancelled:
      name: OrderCancelled
      title: Order cancelled
      contentType: application/json
      payload:
        type: object
```

Noncompliant Code Example (AsyncAPI 3)
--------------------------------------

```yaml
asyncapi: 3.0.0
info:
  title: Orders Events
  version: 1.0.0
channels:
  orderCreated:
    address: orders.created
    messages:
      orderCreated: # Noncompliant {{AAR035: Message should have a 'title' field.}}
        name: OrderCreated
        contentType: application/json
        payload:
          type: object
operations:
  publishOrderCreated:
    action: send
    channel:
      $ref: '#/channels/orderCreated'
    messages:
      - $ref: '#/channels/orderCreated/messages/orderCreated'
components:
  messages:
    OrderCancelled:
      name: OrderCancelled
      title: "" # Noncompliant {{AAR035: Message should have a 'title' field.}}
      contentType: application/json
      payload:
        type: object
```

Compliant Solution (AsyncAPI 3)
-------------------------------

```yaml
asyncapi: 3.0.0
info:
  title: Orders Events
  version: 1.0.0
channels:
  orderCreated:
    address: orders.created
    messages:
      orderCreated:
        name: OrderCreated
        title: Order created
        contentType: application/json
        payload:
          type: object
operations:
  publishOrderCreated:
    action: send
    channel:
      $ref: '#/channels/orderCreated'
    messages:
      - $ref: '#/channels/orderCreated/messages/orderCreated'
components:
  messages:
    OrderCancelled:
      name: OrderCancelled
      title: Order cancelled
      contentType: application/json
      payload:
        type: object
```
